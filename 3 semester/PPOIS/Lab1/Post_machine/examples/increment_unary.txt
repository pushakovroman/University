# Пример программы для Машины Поста: прибавление 1 к унарному числу на ленте
# Начальное состояние: три метки подряд (число 3), каретка на первой метке
TAPE: [1]11

# Правила программы:
# 1. Если под кареткой пусто (0) -> перейти к правилу 2; если метка (1) -> перейти к правилу 3
1: ? 2, 3
# 2. Поставить метку (V) в первую свободную секцию справа и перейти к правилу 4
2: V 4
# 3. Сдвинуть каретку вправо (>) и снова проверить секцию (правило 1)
3: > 1
# 4. Останов (!)
4: !
